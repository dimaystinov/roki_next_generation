cmake_minimum_required(VERSION 3.20)
project(roki_openvino_ncs2 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

set(NCS2_LEGACY_SOURCE_DIR "" CACHE PATH "OpenVINO 2022.3.2 source (transport only)")
set(NCS2_OPENVINO_SOURCE_DIR "" CACHE PATH "Source matching the installed modern OpenVINO runtime")
find_package(OpenVINO 2026.0.0 EXACT REQUIRED COMPONENTS Runtime)
find_package(Threads REQUIRED)
if(NOT EXISTS "${NCS2_OPENVINO_SOURCE_DIR}/src/inference/dev_api/openvino/runtime/iplugin.hpp")
    message(FATAL_ERROR "NCS2_OPENVINO_SOURCE_DIR must contain matching developer headers")
endif()
set(legacy "${NCS2_LEGACY_SOURCE_DIR}/src/plugins/intel_myriad/third_party")
if(NOT EXISTS "${legacy}/mvnc/include/mvnc.h")
    message(FATAL_ERROR "NCS2_LEGACY_SOURCE_DIR must be OpenVINO 2022.3.2")
endif()

# Reuse upstream transport build rules, without its old OpenVINO libraries.
function(openvino_developer_export_targets)
endfunction()
function(ov_install_static_lib)
endfunction()
set(XLINK_DIR "${legacy}/XLink")
add_compile_definitions(XLINK_USE_BUS)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_compile_definitions(_GNU_SOURCE)
endif()
add_subdirectory("${legacy}/XLink" transport/XLink)
add_subdirectory("${legacy}/mvnc" transport/mvnc)

add_library(openvino_ncs2_plugin SHARED src/plugin.cpp)
target_compile_definitions(openvino_ncs2_plugin PRIVATE IMPLEMENT_OPENVINO_RUNTIME_PLUGIN)
target_include_directories(openvino_ncs2_plugin PRIVATE
    "${NCS2_OPENVINO_SOURCE_DIR}/src/inference/dev_api"
    "${NCS2_OPENVINO_SOURCE_DIR}/src/core/dev_api"
    "${NCS2_OPENVINO_SOURCE_DIR}/src/common/util/include"
    "${NCS2_OPENVINO_SOURCE_DIR}/src/common/itt/include")
target_link_libraries(openvino_ncs2_plugin PRIVATE mvnc openvino::runtime)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Upstream mvnc/XLink is not safe to unload; let the OS reclaim at exit.
    target_link_options(openvino_ncs2_plugin PRIVATE "-Wl,-z,nodelete")
endif()
include(GNUInstallDirs)
install(TARGETS openvino_ncs2_plugin LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES LICENSE README_RU.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
add_executable(inspect_blob src/inspect_blob.cpp)
include(CTest)
set(NCS2_TEST_PYTHON "" CACHE FILEPATH "Python with modern OpenVINO and numpy, for integration tests")
if(BUILD_TESTING)
    add_executable(blob_tests tests/blob_tests.cpp)
    target_include_directories(blob_tests PRIVATE src)
    add_test(NAME blob_tests COMMAND blob_tests)
    add_library(openvino_ncs2_test_plugin SHARED src/plugin.cpp tests/fake_mvnc.cpp)
    target_compile_definitions(openvino_ncs2_test_plugin PRIVATE IMPLEMENT_OPENVINO_RUNTIME_PLUGIN NCS2_TEST_TRANSPORT)
    get_target_property(plugin_includes openvino_ncs2_plugin INCLUDE_DIRECTORIES)
    target_include_directories(openvino_ncs2_test_plugin PRIVATE ${plugin_includes} "${legacy}/mvnc/include")
    target_link_libraries(openvino_ncs2_test_plugin PRIVATE openvino::runtime)
    if(NCS2_TEST_PYTHON AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME plugin_integration COMMAND "${NCS2_TEST_PYTHON}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_plugin.py" "$<TARGET_FILE:openvino_ncs2_test_plugin>")
    endif()
endif()
